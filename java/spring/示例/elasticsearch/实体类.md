import org.springframework.data.annotation.Id;
import org.springframework.data.elasticsearch.annotations.Document;
import org.springframework.data.elasticsearch.annotations.Field;
import org.springframework.data.elasticsearch.annotations.FieldType;

@Data
@Document(indexName = "book_page")
public class GtzBookPageDocument implements Serializable {
    // FieldType.Keyword表示查询时必须完全相等，不借助分词查询，ID建议设置为Keyword
    @Id
    @Field(type = FieldType.Keyword)
    private String id;
    // FieldType.TEXT表示查询时使用分词查询，analyzer指定索引分词器，searchAnalyzer指定查询分词器
    @Field(type = FieldType.Text, analyzer = "dic_ansj", searchAnalyzer = "dic_ansj")
    private String content;
    @Field(name = "file_md5", type = FieldType.Keyword, index = false)
    private String fileMd5;
    // index=false表示不索引该字段
    @Field(type = FieldType.Integer, index = false)
    private Integer page;
}
