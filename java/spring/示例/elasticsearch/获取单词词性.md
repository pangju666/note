```java
import org.ansj.domain.Term;
import org.ansj.recognition.impl.NatureRecognition;

public static final NatureRecognition NATURE_RECOGNITION = new NatureRecognition();

public static String getWordNatureStr(String str) {
    List<Term> recognition = NATURE_RECOGNITION.recognition(Collections.singletonList(str));
    return recognition.getFirst().getNatureStr();
}
```
