```java
public class AuthenticationInterceptor extends BaseRequestInterceptor {
    private final TokenService tokenService;

    @Override
    public boolean preHandle(@NonNull HttpServletRequest request, @NonNull HttpServletResponse response, @NonNull Object handler) {
       if (handler instanceof HandlerMethod handlerMethod) {
          Class<?> targetClass = handlerMethod.getBeanType();
          Method targetMethod = handlerMethod.getMethod();

          if (Objects.nonNull(targetMethod.getAnnotation(IgnoreLogin.class))) {
             return true;
          }
          if (Objects.isNull(targetClass.getAnnotation(RequiresLogin.class)) && Objects.isNull(targetMethod.getAnnotation(RequiresLogin.class))) {
             return true;
          }

          String token = request.getHeader(HttpHeaders.AUTHORIZATION);
          if (StringUtils.isBlank(token)) {
                ResponseUtils.writeExceptionToResponse(new InvalidTokenException(), response);
             return false;
          }
          if (!RegExUtils.matches(TokenValidator.PATTERN, token)) {
                ResponseUtils.writeExceptionToResponse(new InvalidTokenException(), response);
             return false;
          }
          token = StringUtils.substringAfter(token, ConstantPool.TOKEN_PREFIX);

            AuthenticatedUser authenticatedUser = tokenService.getAuthenticatedUser(token);
            if (Objects.isNull(authenticatedUser)) {
                ResponseUtils.writeExceptionToResponse(new InvalidTokenException(), response);
                return false;
            }
            RequestContextHolder.currentRequestAttributes().setAttribute(AuthenticationUtils.ACCOUNT_ATTRIBUTE_NAME, authenticatedUser, RequestAttributes.SCOPE_REQUEST);
          return true;
       }
       return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) {
        RequestContextHolder.currentRequestAttributes().removeAttribute(AuthenticationUtils.ACCOUNT_ATTRIBUTE_NAME, RequestAttributes.SCOPE_REQUEST);
    }
}
```
