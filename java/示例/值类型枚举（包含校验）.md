@Getter
public enum ValueType {
    STRING("string", "字符串", Objects::nonNull),
    NUMBER("number", "数字", NumberUtils::isCreatable),
    BOOLEAN("boolean", "布尔", value -> Objects.nonNull(BooleanUtils.toBooleanObject(value))),
    OBJECT("object", "JSON", value -> {
        try {
            JsonElement json = JsonUtils.parseString(value);
            return json != JsonNull.INSTANCE && json.isJsonObject();
        } catch (JsonParseException e) {
            return false;
        }
    });

    @EnumValue
    private final String type;
    private final String label;
    private final Predicate<String> predicate;

    ValueType(String type, String label, Predicate<String> predicate) {
        this.type = type;
        this.label = label;
        this.predicate = predicate;
    }
}
