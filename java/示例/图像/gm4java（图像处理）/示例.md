# 进程调用

```java
import org.gm4java.engine.GMConnection;
import org.gm4java.engine.GMException;
import org.gm4java.engine.GMServiceException;
import org.gm4java.engine.support.PooledGMService;
import org.gm4java.im4java.GMOperation;

public String generate(Supplier<GMOperation> supplier, Consumer<Exception> consumer) {
    GMConnection connection = null;
    try {
       connection = pooledGMService.getConnection();
       GMOperation operation = supplier.get();
       return connection.execute(operation.toString());
    } catch (IOException | GMServiceException | GMException e) {
       consumer.accept(e);
       throw new ImageDamageException();
    } finally {
       if (Objects.nonNull(connection)) {
          try {
             connection.close();
          } catch (GMServiceException e) {
             log.error("GraphicsMagick进程关闭失败");
          }
       }
    }
}
```

# 生成缩略图

```java
if (width < sourceImage.getWidth() && height < sourceImage.getHeight()) {
    return 原图;
}

generate(() -> {
    GMOperation operation = new GMOperation();
    if (watermark) {
       operation.addRawArg("composite");
       // 建议水印缩放率为0.2
       operation.geometry((int) (expectWidth * watermarkScale),
          (int) (expectHeight * watermarkScale));
        // 是否需要旋转（有些图像存在orientation属性）
       if (isRotate()) {
           // 定义水印方向
          operation.gravity(GMOperation.Gravity.NorthEast);
          operation.rotate(-90d);
       } else {
          operation.gravity(GMOperation.Gravity.SouthEast);
       }
       // 建议水印不透明度为70
       operation.addRawArg("-dissolve " + watermarkOpacity);
       operation.addImage(watermarkFile);
    } else {
       operation.addRawArg("convert");
    }
    operation.addImage(sourceImage);
    // 判断原图是否大于缩略图尺寸，大于则需要缩放
    if (!finalTmpOriginalWatermark) {
       if (force) {
          operation.resize(expectWidth, expectHeight, '!');
       } else {
          operation.resize(expectWidth, expectHeight);
       }
    }
     // 是否需要旋转（有些图像存在orientation属性）
    if (!isRotate()) {
       // 移除元属性（例如：orientation）
       operation.stripProfiles();
    }
    // 图像质量建议为80
    operation.quality(imageQuality);
    operation.addImage(outputFile);
    return operation;
}, e -> log.error("图片缩略图生成失败", e));
```
