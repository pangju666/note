# 全景图切片

```java
import org.apache.commons.exec.CommandLine;
import org.apache.commons.exec.DefaultExecutor;
import org.apache.commons.exec.Executor;

Executor executor = DefaultExecutor.builder().get();

CommandLine commandLine = new CommandLine(krpanoToolsConfig.getPath());
commandLine.addArgument(krpanoToolsConfig.getSphereToCube().getName());
commandLine.addArgument(krpanoToolsConfig.getSphereToCube().getMode());
commandLine.addArgument(inputPath);
commandLine.addArgument(outputPath);
for (String extraArgument : krpanoToolsConfig.getSphereToCube().getExtraArguments()) {
    commandLine.addArgument(extraArgument);
}
try {
    executor.execute(commandLine);
} catch (IOException e) {
    log.error("全景图切片失败", e);
    throw new ServiceException("全景图切片失败");
}
```
