# 依赖包

```xml
<dependency>
    <groupId>org.apache.commons</groupId>
    <artifactId>commons-exec</artifactId>
</dependency>
```

```
krpano-tools:
  path: /usr/local/bin/krpanotools
  sphere-to-cube:
    name: sphere2cube
    mode: CUBE
    extra-arguments:
      - '-jpegoptimize'
      - '-jpegprogressive'
      - '-q'
```

```java
@Data
@SpringBootConfiguration
@ConfigurationProperties(prefix = "krpano-tools")
public class KrpanoToolsConfig {
    private String path;
    private SphereToCube sphereToCube;

    @Data
    public static class SphereToCube {
       private String name;
       private String mode;
       private String[] extraArguments;
    }
}
```
