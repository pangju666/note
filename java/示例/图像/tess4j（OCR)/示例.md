# 图像OCR

```java
import net.sourceforge.tess4j.ITesseract;
import net.sourceforge.tess4j.Tesseract;
import net.sourceforge.tess4j.TesseractException;
import java.awt.*;

Map<String, ITesseract> tesseractMap = new HashMap<>(tess4jConfig.getLanguages().size());
for (String language : tess4jConfig.getLanguages()) {
    ITesseract tesseract = new Tesseract();
    tesseract.setDatapath(tess4jConfig.getDataPath());
    tesseract.setLanguage(language);
    this.tesseractMap.put(language, tesseract);
}

private void sortRectangles(List<Rectangle> rectangles) {
    rectangles.sort((left, right) -> {
       int result = left.x - right.x;
       if (result == 0) {
          result = left.y - right.y;
          if (result == 0) {
             result = left.width - right.width;
             if (result == 0) {
                result = left.height - right.height;
             }
          }
       }
       return result;
    });
}

try {
    ITesseract iTesseract = this.tesseractMap.get(language);
    if (Objects.isNull(iTesseract)) {
       throw new ServiceException("不支持该语言");
    }

    if (CollectionUtils.isNotEmpty(rectangles)) {
       return iTesseract.doOCR(originalFile, rectangles);
    }
    return iTesseract.doOCR(originalFile);
} catch (TesseractException e) {
    log.error("图片OCR失败", e);
    throw new ServiceException("图片ocr失败");
} finally {
    FileUtils.deleteTmpFiles(originalFile);
}
```
