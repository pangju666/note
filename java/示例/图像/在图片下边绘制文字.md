public BufferedImage addTextToImage(BufferedImage image, Font font, String text) {
    Graphics2D graphics = image.createGraphics();

    graphics.setFont(font);
    FontMetrics metrics = graphics.getFontMetrics();

    // 计算文字的宽度和高度
    int textWidth = metrics.stringWidth(text);
    // 加入一些额外的空间作为padding
    int textHeight = metrics.getAscent() + metrics.getDescent() + metrics.getHeight() / 4;

    // 根据文字宽度和原始图像宽度确定新的图像宽度
    // 增加一些额外的空间作为padding
    int newImageWidth = Math.max(image.getWidth(), textWidth + 40);
    int newImageHeight = image.getHeight() + textHeight;

    // 创建一个新的BufferedImage对象，宽度和高度都增加以容纳文字
    BufferedImage newImage = new BufferedImage(newImageWidth, newImageHeight, BufferedImage.TYPE_INT_ARGB);
    // 释放之前的Graphics2D资源
    graphics.dispose();

    graphics = newImage.createGraphics();
    graphics.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
    // 将原始图像绘制到新图像上，并居中
    graphics.drawImage(image, (newImageWidth - image.getWidth()) / 2, 0, null);

    // 计算文字渲染坐标，并水平居中
    int x = (newImageWidth - textWidth) / 2;
    int y = newImage.getHeight() - textHeight + metrics.getAscent();
    // 设置字体和颜色
    graphics.setFont(font);
    graphics.setColor(Color.WHITE);
    // 绘制文字
    graphics.drawString(text, x, y);

    // 释放资源
    graphics.dispose();

    return newImage;
}
