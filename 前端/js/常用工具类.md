import Crypto from "crypto-js";
import SparkMD5 from "spark-md5";
import {
  ArrayUtils,
  DateFormatUtils,
  ObjectUtils,
  StringUtils,
} from "pangju-utils";

const FILE_MD5_SLICE_SIZE = 1024 * 1024 * 4;
export const IDENTIFIER_REGEX = /^[a-zA-Z_][a-zA-Z0-9-_]*$/;
export const MAX_FILE_UPLOAD_SIZE = 1024 * 1024 * 1024 * 5;
export const MAX_KML_FILE_UPLOAD_SIZE = 1024 * 1024 * 100;
export const DEFAULT_TEXT = "暂无";

export function convertDecimalToDMS(decimal) {
  const degrees = Math.floor(decimal);
  const minutes = Math.floor((decimal - degrees) * 60);
  const seconds = ((decimal - degrees - minutes / 60) * 3600).toFixed(2);
  return `${degrees}°${minutes}'${seconds}"`;
}

export async function getResult(value) {
  if (value instanceof Function) {
    const result = value();
    if (result instanceof Promise) {
      return await result;
    } else {
      return result;
    }
  } else {
    if (value instanceof Promise) {
      return await value;
    } else {
      return value;
    }
  }
}

export function getValueType(value, nullType = "string") {
  if (ObjectUtils.isNull(value)) {
    return nullType;
  } else if (value === true || value === false) {
    return "boolean";
  } else if (!isNaN(value) && typeof value === "number") {
    return "number";
  } else if (typeof value === "object") {
    return "object";
  } else {
    return "string";
  }
}

export function computeApiSignature(appId, secretKey, url, timestamp) {
  if (ObjectUtils.nonNull(timestamp)) {
    return Crypto.SHA1(
      `${appId}&${secretKey}&${encodeURIComponent(url)}&${timestamp}`,
    );
  }
  return Crypto.SHA1(`${appId}&${secretKey}&${encodeURIComponent(url)}`);
}

export function sleep(delay) {
  return new Promise((resolve) => setTimeout(resolve, delay));
}

export function copyToClipboard(text, successCallback, failureCallback) {
  if (!navigator.clipboard) {
    const textArea = document.createElement("textarea");
    textArea.value = text;
    document.body.appendChild(textArea);
    textArea.focus();
    textArea.select();
    document.execCommand("copy");
    document.body.removeChild(textArea);
    successCallback();
  } else {
    navigator.clipboard.writeText(text).then(successCallback, failureCallback);
  }
}

export async function calcFileMd5(file) {
  const spark = new SparkMD5();
  if (file?.size >= FILE_MD5_SLICE_SIZE) {
    const middle = file.size >> 1;
    const head = await file.slice(0, 1000).arrayBuffer();
    const central = await file.slice(middle - 500, middle + 500).arrayBuffer();
    const foot = await file.slice(file.size - 1000, file.size).arrayBuffer();
    const headList = new Uint8Array(head);
    const centralList = new Uint8Array(central);
    const footList = new Uint8Array(foot);
    spark.append(headList);
    spark.append(centralList);
    spark.append(footList);
  } else {
    const all = await file.arrayBuffer();
    const allList = new Uint8Array(all);
    spark.append(allList);
  }
  let md5 = spark.end();
  spark.destroy();
  return md5;
}

export function downloadFile(url, filename) {
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  a.target = "_blank";
  a.click();
  a.remove();
}

export function defaultTextIfBlank(text, defaultText = DEFAULT_TEXT) {
  return StringUtils.defaultIfBlank(text, defaultText);
}

export function formatDateTime(date) {
  if (ObjectUtils.isNull(date)) {
    return DEFAULT_TEXT;
  }
  return DateFormatUtils.format(date, DateFormatUtils.DATE_TIME_FORMAT);
}

export function formatDate(date) {
  if (ObjectUtils.isNull(date)) {
    return DEFAULT_TEXT;
  }
  return DateFormatUtils.format(date, DateFormatUtils.DATE_FORMAT);
}

export function formatFileSize(fileSize) {
  const num = 1024.0;
  if (fileSize < num) {
    return fileSize + "B";
  }
  if (fileSize < Math.pow(num, 2)) {
    return (fileSize / num).toFixed(2) + "KB";
  }
  if (fileSize < Math.pow(num, 3)) {
    return (fileSize / Math.pow(num, 2)).toFixed(2) + "MB";
  }
  if (fileSize < Math.pow(num, 4)) {
    return (fileSize / Math.pow(num, 3)).toFixed(2) + "G";
  }
  return (fileSize / Math.pow(num, 4)).toFixed(2) + "T";
}

export function toTree(
  source,
  props = {
    id: "id",
    parentId: "parentId",
    children: "children",
  },
  root = null,
  convertFunc = null,
) {
  if (ArrayUtils.isEmpty(source)) {
    return [];
  }

  const tree = [];
  const map = new Map(
    source.map((item) => {
      const node = item;
      node[props.children] = [];
      return [item[props.id], node];
    }),
  );
  for (let item of source) {
    if (item[props.parentId] === root) {
      tree.push(item);
    } else {
      const parent = map.get(item[props.parentId]);
      if (ObjectUtils.nonNull(parent)) {
        parent.children.push(item);
      }
    }
    if (ObjectUtils.nonNull(convertFunc)) {
      convertFunc(item);
    }
  }
  return tree;
}

export function getFileSuffix(name) {
  if (!name) {
    return null;
  }
  let split = name.split(".");
  if (split.length < 2) {
    return null;
  }
  return split[split.length - 1];
}
