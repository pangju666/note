import { h } from "vue";
import { NButton, NIcon, NImage, NPopconfirm, NTooltip } from "naive-ui";
import SvgIcon from "@/components/common/SvgIcon.vue";
import { downloadFile, getResult } from "@/utils/utils.js";
import { FileApi } from "@/apis/file/FileApi.js";
import { StringUtils } from "pangju-utils";

export function renderImage(imageMd5, defaultImage, width = 40) {
  return h(NImage, {
    width: width,
    objectFit: "scale-down",
    lazy: true,
    fallbackSrc: "https://assets.changtech.cn/image/image-error.jpg",
    src: StringUtils.isBlank(imageMd5)
      ? defaultImage
      : FileApi.getImagePreviewUrl(imageMd5),
    previewDisabled: StringUtils.isBlank(imageMd5),
    previewSrc: StringUtils.isBlank(imageMd5)
      ? FileApi.getDownloadUrl(imageMd5)
      : undefined,
  });
}

export function renderFileIcon(
  fileMd5,
  mimeType,
  size = 40,
  objectFit = "scale-down",
) {
  if (mimeType.startsWith("image/")) {
    return h(NImage, {
      width: size,
      objectFit: objectFit,
      lazy: true,
      fallbackSrc: "https://assets.changtech.cn/image/image-error.jpg",
      src: FileApi.getImagePreviewUrl(fileMd5),
      previewSrc: FileApi.getDownloadUrl(fileMd5),
    });
  } else if (mimeType.startsWith("text/")) {
    return renderIcon("icon-wenbenwendang-txt", size);
  } else if (mimeType.startsWith("video/")) {
    return renderIcon("icon-shipin", size);
  } else if (mimeType.startsWith("audio/")) {
    return renderIcon("icon-yinpin", size);
  } else if (
    mimeType === "application/x-zip-compressed" ||
    mimeType === "application/x-7z-compressed" ||
    mimeType === "application/x-compressed" ||
    mimeType === "application/gzip"
  ) {
    return renderIcon("icon-yasuobao", size);
  } else if (
    mimeType ===
      "application/vnd.openxmlformats-officedocument.wordprocessingml.document" ||
    mimeType === "application/msword"
  ) {
    return renderIcon("icon-office-word", size);
  } else if (
    mimeType ===
      "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet" ||
    mimeType === "application/vnd.ms-excel"
  ) {
    return renderIcon("icon-office-excel", size);
  } else if (
    mimeType ===
      "application/vnd.openxmlformats-officedocument.presentationml.presentation" ||
    mimeType === "application/vnd.ms-powerpoint"
  ) {
    return renderIcon("icon-office-pptx", size);
  } else if (mimeType === "application/pdf") {
    return renderIcon("icon-pdf", size);
  } else {
    return renderIcon("icon-file-unknown-fill", size);
  }
}

export function renderDownloadButton(
  filename,
  fileUrl,
  buttonClass = "",
  buttonText = "下载",
) {
  return h(
    NButton,
    {
      class: buttonClass,
      secondary: true,
      type: "success",
      onClick: async () => {
        downloadFile(await getResult(fileUrl), filename);
      },
    },
    buttonText,
  );
}

export function renderPopConfirm(
  callback,
  triggerText,
  buttonClass = "",
  buttonText = "删除",
  buttonType = "error",
  buttonDisabled = false,
) {
  return h(
    NPopconfirm,
    {
      onPositiveClick: () => {
        callback();
      },
    },
    {
      trigger: () =>
        h(
          NButton,
          {
            class: buttonClass,
            disabled: buttonDisabled,
            secondary: true,
            type: buttonType,
          },
          buttonText,
        ),
      default: () => triggerText,
    },
  );
}

export function renderIcon(icon) {
  return () => h(NIcon, null, { default: () => h(icon) });
}

export function renderSvgIcon(icon, size) {
  return h(
    NIcon,
    { size },
    {
      default: () => {
        if (typeof icon === "string") {
          return h(SvgIcon, { icon });
        }
        return h(icon);
      },
    },
  );
}

export function renderToolTipOption({ node, option }) {
  return renderToolTip(node, option.label, "right");
}

export function renderToolTip(node, text, placement = "bottom") {
  return h(
    NTooltip,
    {
      placement: placement,
    },
    {
      trigger: () => node,
      default: () => text,
    },
  );
}
