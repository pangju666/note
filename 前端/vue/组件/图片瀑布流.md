<template>
  <div style="width: 100%" ref="waterfallRef">
    <waterfall
      v-if="jades.length > 0"
      class="jade-waterfall"
      :data="jades"
      width="300px"
      gap="10px"
      delay
    >
      <template #default="item">
        <div
          class="jade-card cursor-pointer"
          @click="$router.push(`/view/${item.id}`)"
        >
          <div class="jade-image">
            <img
              v-if="item.imageUrl"
              alt=""
              :src="item.imageUrl"
              @load="onImageLoad"
            />
            <img
              v-else
              alt=""
              src="@/assets/images/empty-image.jpg"
              @load="onImageLoad"
            />
          </div>
          <div class="jade-content">
            <div class="overflow-text" :title="item.utensil">
              {{ item.utensil }}
            </div>
            <div class="jade-subtitle">
              <div
                class="overflow-text"
                :title="item.museum"
                style="width: 60%"
              >
                {{ item.museum }}
              </div>
              <div
                class="overflow-text h-right"
                :title="item.collectionNumber"
                style="width: 40%"
              >
                （{{ item.collectionNumber }}）
              </div>
            </div>
          </div>
        </div>
      </template>
    </waterfall>
    <div v-show="!loading" class="pull-icon">
      <img
        src="@/assets/images/pull.png"
        class="cursor-pointer"
        @click="onClickLoad(condition)"
      />
    </div>
    <div v-show="loading" class="loading-icon">
      <img src="@/assets/images/loading.gif" />
    </div>
  </div>
</template>

<script setup>
import { ref } from "vue";
import { JadeApi } from "@/api/JadeApi";
import { Toast } from "@/assets/js/Toast";
import Waterfall from "kuan-vue-waterfall";
import { StringUtils } from "pangju-utils";
import { getSignatureUrl } from "@/assets/js/ApiSignature";

defineProps({
  condition: {
    type: Object,
    default: () => {},
  },
});

let loading = ref(false);
let hasData = false;
let currentPage = 1;
let pageSize = 30;
const jades = ref([]);
let imageRenderCount = 0;

const onClickLoad = (condition) => {
  loadNextPage(condition, false);
};

const loadNextPage = async (condition, refresh) => {
  if (loading.value) {
    return;
  }

  loading.value = true;
  if (refresh) {
    imageRenderCount = 0;
    currentPage = 1;
    jades.value = [];
    hasData = false;
  }

  const res = await JadeApi.page(currentPage, pageSize, condition);
  if (currentPage >= res.pages && hasData) {
    Toast.info("没有更多了");
    loading.value = false;
    return;
  }
  res.records.forEach((jade) => {
    if (StringUtils.isNotEmpty(jade.coverMd5)) {
      jade.imageUrl = getSignatureUrl(
        `${process.env.VUE_APP_BASE_FILE_URL}/image/preview/${jade.coverMd5}?placeholder=true`
      );
    }
    jade.key = jade.id;
  });
  if (refresh) {
    jades.value = res.records;
  } else {
    jades.value = [...jades.value, ...res.records];
  }
  hasData = true;
  ++currentPage;
  setTimeout(() => {
    loading.value = false;
  }, 3000);
};

const onImageLoad = () => {
  ++imageRenderCount;
  if (imageRenderCount === jades.value.length) {
    loading.value = false;
  }
};

defineExpose({ loadNextPage });
</script>
