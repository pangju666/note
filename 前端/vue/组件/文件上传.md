```javascript
<script setup>
import { NButton, useMessage } from "naive-ui";
import { computed, ref } from "vue";
import { getToken } from "@/utils/token.js";
import {
  calcFileMd5,
  downloadFile,
  formatFileSize,
  MAX_FILE_UPLOAD_SIZE,
} from "@/utils/utils.js";
import { ObjectUtils, StringUtils } from "pangju-utils";
import { AddOutline, CloudUploadOutline } from "@vicons/ionicons5";
import { FileApi } from "@/apis/file/FileApi.js";

const message = useMessage();

const props = defineProps({
  buttonClass: {
    type: String,
    default: "",
  },
  imageCardMode: {
    type: Boolean,
    default: false,
  },
  accept: {
    type: String,
    default: undefined,
  },
  uploadUrl: {
    type: [String, Function],
    required: true,
  },
  onBeforeUpload: {
    type: Function,
    default: null,
  },
  onUploadSuccess: {
    type: Function,
    required: true,
  },
  max: {
    type: Number,
    default: undefined,
  },
  multiple: {
    type: Boolean,
    default: false,
  },
  buttonText: {
    type: String,
    default: "上传",
  },
  maxUploadSize: {
    type: Number,
    default: MAX_FILE_UPLOAD_SIZE,
  },
  onRemove: {
    type: Function,
    default: null,
  },
});

defineEmits(["upload-change", "update-file-list"]);

const uploadRef = ref();
const uploadFileMd5 = ref(null);
const uploadLoading = ref(false);

const fileUploadUrl = computed(() => {
  if (ObjectUtils.isNull(props.uploadUrl)) {
    return null;
  }
  if (typeof props.uploadUrl === "string") {
    return props.uploadUrl;
  }
  return props.uploadUrl(uploadFileMd5.value);
});
const requestHeaders = computed(() => ({
  Authorization: getToken(),
}));
const maxUploadSizeLabel = computed(() => formatFileSize(props.maxUploadSize));
const listType = computed(() =>
  props.imageCardMode ? "image-card" : undefined,
);

const createImage = (fileMd5, mimeType = null) => {
  return {
    status: "finished",
    type: mimeType ?? "image/",
    url: FileApi.getDownloadUrl(fileMd5),
    thumbnailUrl: FileApi.getImagePreviewUrl(fileMd5),
  };
};
const createThumbnailUrl = (file, fileInfo) => {
  if (StringUtils.isNotBlank(fileInfo.thumbnailUrl)) {
    return fileInfo.thumbnailUrl;
  }
  if (StringUtils.isBlank(fileInfo.url)) {
    return URL.createObjectURL(file);
  }
  return FileApi.getImagePreviewUrl(uploadFileMd5.value);
};
const shouldUseThumbnailUrl = (file) => {
  return StringUtils.isNotBlank(file.type) && file.type.startsWith("image/");
};
const isUploadFailure = (xhr) => {
  return xhr.status !== 200 || xhr.response.code < 0;
};

const onBeforeUpload = async ({ file }) => {
  if (file.file.size > props.maxUploadSize) {
    message.warning(`最大可上传${maxUploadSizeLabel.value}文件`);
    return Promise.resolve(false);
  }
  if (ObjectUtils.nonNull(props.onBeforeUpload)) {
    const result = props.onBeforeUpload(file);
    if (typeof result === "boolean" && !result) {
      return Promise.resolve(false);
    }
    if (!(await result)) {
      return Promise.resolve(false);
    }
  }
  uploadLoading.value = true;
  uploadFileMd5.value = await calcFileMd5(file.file);
};
const onUploadError = ({ event }) => {
  message.error(event.target.response.message);
  uploadLoading.value = false;
  uploadRef.value.clear();
};
const onUploadFinish = ({ file }) => {
  uploadLoading.value = false;
  uploadRef.value.clear();
  props.onUploadSuccess(uploadFileMd5.value, file);
};
const onFileRemove = ({ file }) => {
  if (ObjectUtils.nonNull(props.onRemove)) {
    return props.onRemove(file);
  }
  return true;
};
const onDownload = (file) => {
  downloadFile(file.url, file.name);
};

defineExpose({ createImage });
</script>

<template>
  <n-upload
    ref="uploadRef"
    :action="fileUploadUrl"
    :headers="requestHeaders"
    :accept="accept"
    :create-thumbnail-url="createThumbnailUrl"
    :is-error-state="isUploadFailure"
    :list-type="listType"
    :max="max"
    :multiple="multiple"
    :on-before-upload="onBeforeUpload"
    :on-remove="onFileRemove"
    response-type="json"
    :should-use-thumbnail-url="shouldUseThumbnailUrl"
    :show-file-list="imageCardMode"
    show-download-button
    @error="onUploadError"
    @finish="onUploadFinish"
    @download="onDownload"
    @change="$emit('upload-change', $event)"
  >
    <n-icon v-if="imageCardMode" :size="30">
      <AddOutline />
    </n-icon>
    <n-button
      v-else
      :class="buttonClass"
      :loading="uploadLoading"
      secondary
      type="info"
    >
      <template #icon>
        <n-icon>
          <cloud-upload-outline />
        </n-icon>
      </template>
      {{ buttonText }}
    </n-button>
  </n-upload>
</template>
```
