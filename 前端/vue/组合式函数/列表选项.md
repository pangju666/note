```javascript
import { ref } from "vue";
import { ObjectUtils } from "pangju-utils";
import { getResult } from "@/utils/utils.js";

export function useOptions(
  data,
  mapFunc = (item) => ({ label: item, value: item }),
  groupFunc = null,
) {
  const loading = ref(false);
  const options = ref([]);

  const getOptions = async () => {
    loading.value = true;
    options.value = (await getResult(data)).map((item) => mapFunc(item));
    if (ObjectUtils.nonNull(groupFunc)) {
      options.value = groupFunc(options.value);
    }
    loading.value = false;
  };

  return { loading, options, getOptions };
}
```
