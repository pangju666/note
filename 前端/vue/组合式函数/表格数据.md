import { reactive, ref } from "vue";

export function useTableData(request) {
  const loading = ref(true);
  const pagination = reactive({
    page: 1,
    pageSize: 20,
    showSizePicker: true,
    showQuickJumper: true,
    pageSizes: [20, 50, 100, 150, 200],
    pageCount: 1,
    onUpdatePage: (page) => {
      pagination.page = page;
      getData(page, pagination.pageSize);
    },
    onUpdatePageSize: (pageSize) => {
      pagination.pageSize = pageSize;
      pagination.page = 1;
      getData(1, pageSize);
    },
  });
  const data = ref([]);

  const getData = async (
    page = pagination.page,
    size = pagination.pageSize,
  ) => {
    loading.value = true;
    const res = await request(page, size);
    pagination.pageCount = res.pages;
    data.value = res.records;
    loading.value = false;

    if (data.value.length === 0 && pagination.page !== 1) {
      pagination.page = pagination.page - 1;
      getData();
    }
  };

  const resetData = () => {
    data.value = [];
    pagination.page = 1;
    getData();
  };

  return { loading, pagination, data, getData, resetData };
}
