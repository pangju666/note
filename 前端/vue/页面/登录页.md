<script setup>
import VerificationCode from "@/components/common/VerificationCode.vue";
import { ObjectUtils, StringUtils } from "pangju-utils";
import { ref } from "vue";
import { JSEncrypt } from "jsencrypt";
import { getAuthenticateKey, login } from "@/apis/authenticate.js";
import { useNotification } from "naive-ui";
import { redirectToHome, setToken } from "@/utils/token.js";

const notification = useNotification();

const formRef = ref();
let codeBoxRef = ref();

const model = ref({
  username: null,
  password: null,
  validateCode: null,
});
let validateCode = ref(null);

const rules = {
  username: {
    required: true,
    message: "请输入用户名",
    trigger: "blur",
  },
  password: {
    required: true,
    message: "请输入密码",
    trigger: "blur",
  },
  validateCode: [
    {
      required: true,
      message: "请输入验证码",
      trigger: "blur",
    },
    {
      validator(rule, value) {
        if (StringUtils.getLength(value) !== 4) {
          return new Error("验证码长度不正确");
        }
        return true;
      },
      trigger: "blur",
    },
    {
      validator(rule, value) {
        if (!StringUtils.equalsIgnoreCase(value, validateCode.value)) {
          return new Error("验证码不正确");
        }
        return true;
      },
      trigger: "blur",
    },
  ],
};

const handleLogin = () => {
  formRef.value.validate(async (errors) => {
    if (ObjectUtils.nonNull(errors)) {
      codeBoxRef.value.changeCode();
    } else {
      try {
        const key = await getAuthenticateKey();
        const encrypt = new JSEncrypt();
        encrypt.setPublicKey(key);
        const encryptPassword = encrypt.encrypt(model.value.password);
        const res = await login(model.value.username, encryptPassword);
        setToken(res.username, res.roles, res.token);
        redirectToHome();
      } catch {
        codeBoxRef.value.changeCode();
        notification.error({
          content: "登录失败，请联系超级管理员处理",
        });
      }
    }
  });
};
</script>

<template>
  <div class="login-view full-size">
    <div class="login-card">
      <div class="logo">
        <img src="
      </div>
      <n-form ref="formRef" :model="model" :rules="rules">
        <n-form-item label="用户名" path="username" first>
          <n-input v-model:value="model.username" />
        </n-form-item>
        <n-form-item label="密码" path="password" first>
          <n-input
            v-model:value="model.password"
            show-password-on="mousedown"
            type="password"
          />
        </n-form-item>
        <div class="flex-space-between">
          <n-form-item first label="验证码" path="validateCode">
            <n-input
              v-model:value="model.validateCode"
              @keydown.enter="handleLogin"
            />
          </n-form-item>
          <verification-code
            ref="codeBoxRef"
            :content-height="34"
            :content-width="120"
            class="mt-10"
            @change="validateCode = $event"
          />
        </div>
        <div class="flex-right">
          <n-button type="primary" @click="handleLogin">登录</n-button>
        </div>
      </n-form>
    </div>
  </div>
</template>
