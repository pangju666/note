```javascript
<script setup>
import { h, onActivated, onMounted, ref } from "vue";
import { FileApi } from "@/apis/file/FileApi.js";
import { ObjectUtils, StringUtils } from "pangju-utils";
import {
  copyToClipboard,
  downloadFile,
  formatDateTime,
  formatFileSize,
  getResult,
} from "@/utils/utils.js";
import { NButton, NPopselect, useMessage } from "naive-ui";
import FileReferenceTableModal from "@/components/custom/file/FileReferenceTableModal.vue";
import FileInfoFormModal from "@/components/custom/file/FileInfoFormModal.vue";
import {
  renderFileIcon,
  renderPopConfirm,
  renderToolTipOption,
} from "@/utils/render.js";
import { CompressApi } from "@/apis/file/CompressApi.js";
import FileUpload from "@/components/custom/common/FileUpload.vue";
import { useTableData } from "@/hooks/TableData.js";
import { useOptions } from "@/hooks/Options.js";
import SearchForm from "@/components/common/SearchForm.vue";
import { BriefcaseOutline } from "@vicons/ionicons5";
import { SphereApi } from "@/apis/file/SphereApi.js";
import { OcrApi } from "@/apis/file/OcrApi.js";

const message = useMessage();

onActivated(() => {
  getData();
});

onMounted(() => {
  getFileTypeOptions();
  getMimeTypeOptions();
});

const columns = [
  {
    type: "selection",
  },
  {
    title: "ID",
    key: "id",
    width: 100,
    sorter: "default",
  },
  {
    title: "预览图",
    key: "icon",
    width: 70,
    render: (rowData) => renderFileIcon(rowData.md5, rowData.mimeType),
  },
  {
    title: "文件名",
    key: "name",
    sorter: "default",
    resizable: true,
    ellipsis: {
      tooltip: true,
    },
    render: (rowData) => rowData.name + "." + rowData.extension,
  },
  {
    title: "文件MD5",
    key: "md5",
    width: 300,
  },
  {
    title: "文件类型",
    key: "mimeType",
    sorter: "default",
    width: 250,
    resizable: true,
    ellipsis: {
      tooltip: true,
    },
  },
  {
    title: "文件大小",
    key: "size",
    sorter: "default",
    width: 100,
    render: (rowData) => formatFileSize(rowData.size),
  },
  {
    title: "上传时间",
    key: "createTime",
    sorter: "default",
    width: 200,
    render: (rowData) => formatDateTime(rowData.createTime),
  },
  {
    title: "操作",
    align: "left",
    width: 220,
    fixed: "right",
    render: (rowData) => {
      return h(
        "div",
        {
          class: "display-flex",
        },
        [
          h(
            NButton,
            {
              secondary: true,
              class: "mr-10",
              type: "info",
              onClick: () => {
                currentFileId.value = rowData.id;
                fileInfoModalVisible.value = true;
              },
            },
            "编辑",
          ),
          renderPopConfirm(
            () => {
              FileApi.remove(rowData.id)
                .then(() => getData())
                .catch(() => {});
            },
            "是否确认删除该文件？",
            "mr-10",
          ),
          h(
            NPopselect,
            {
              scrollable: true,
              options: [
                {
                  label: "预览",
                  value: "preview",
                  type: "info",
                  onClick: () =>
                    window.open(
                      `${import.meta.env.VITE_BASE_FILE_VIEWER_URL}/#/${rowData.md5}`,
                      "_blank",
                    ),
                },
                {
                  label: "引用信息",
                  value: "reference",
                  type: "primary",
                  onClick: () => {
                    referenceFileMd5.value = rowData.md5;
                    referenceModalVisible.value = true;
                  },
                },
                {
                  label: "复制下载链接",
                  value: "download-url",
                  type: "success",
                  onClick: async () =>
                    copyToClipboard(
                      FileApi.getDownloadUrl(rowData.md5, true),
                      () => message.success("复制成功"),
                      () => message.error("复制失败"),
                    ),
                },
                {
                  label: "下载",
                  value: "download",
                  type: "success",
                  onClick: async () =>
                    downloadFile(
                      await getResult(
                        FileApi.getDownloadUrl(rowData.md5, true),
                      ),
                      rowData.name + "." + rowData.extension,
                    ),
                },
                {
                  label: "图像OCR",
                  value: "ocr",
                  type: "error",
                  loading: ocrLoading,
                  disabled:
                    !rowData.mimeType.startsWith("image/") &&
                    rowData.mimeType !== "application/pdf",
                  onClick: async () => {
                    ocrLoading.value = true;
                    OcrApi.ocrImage(rowData.md5).then(() => {
                      ocrLoading.value = false;
                      message.success("图像OCR完成");
                    });
                  },
                },
                {
                  label: "全景切片",
                  value: "sphere",
                  type: "warning",
                  disabled: !rowData.mimeType.startsWith("image/"),
                  onClick: async () => {
                    SphereApi.generateCube(rowData.md5).then(() => {
                      message.success("全景切片任务已提交");
                    });
                  },
                },
              ],
              renderLabel: (option) =>
                h(
                  NButton,
                  {
                    type: option.type,
                    loading: option?.loading?.value ?? false,
                    disabled: option?.disabled ?? false,
                    text: true,
                    onClick: option?.onClick,
                  },
                  option.label,
                ),
            },
            h(NButton, {}, "更多"),
          ),
        ],
      );
    },
  },
];
const { loading, pagination, data, getData, resetData } = useTableData(
  (page, size) =>
    FileApi.page(
      page,
      size,
      searchModel.value.keyword,
      searchModel.value.extension,
      searchModel.value.mimeType,
      ObjectUtils.isNull(searchModel.value.mimeType)
        ? searchModel.value.type
        : null,
      sortType.value,
      isAsc.value,
    ),
);
const ocrLoading = ref(false);
const selectionFileMd5s = ref([]);
const sortType = ref(null);
const isAsc = ref(null);

const onTableSorterChange = (sorter) => {
  if (sorter) {
    switch (sorter.columnKey) {
      case "mimeType":
        sortType.value = "MIME_TYPE";
        break;
      case "createTime":
        sortType.value = "create_time";
        break;
      default:
        sortType.value = sorter.columnKey;
        break;
    }
    isAsc.value = sorter.order !== "descend";
  } else {
    sortType.value = null;
    isAsc.value = null;
  }
  resetData();
};

const {
  loading: fileTypeLoading,
  options: fileTypeOptions,
  getOptions: getFileTypeOptions,
} = useOptions(FileApi.getFileTypes(), (item) => ({
  label: item.label,
  value: item.value,
}));

const {
  loading: mimeTypesLoading,
  options: mimeTypeOptions,
  getOptions: getMimeTypeOptions,
} = useOptions(
  FileApi.getMimeTypes,
  (item) => ({ label: item, value: item }),
  (options) => {
    const fileMimeTypesGroup = new Map();
    for (let option of options) {
      const groupName = StringUtils.substringBefore(option.label, "/");
      let group = fileMimeTypesGroup.get(groupName);
      if (ObjectUtils.isNull(group)) {
        group = {
          type: "group",
          label: groupName,
          key: groupName,
          children: [],
        };
        fileMimeTypesGroup.set(groupName, group);
      }
      group.children.push(option);
    }
    return Array.from(fileMimeTypesGroup.values());
  },
);

const searchModel = ref({
  keyword: null,
  extension: null,
  mimeType: null,
  type: null,
});

const onClearSearchForm = () => {
  searchModel.value = {
    keyword: null,
    mimeType: null,
    type: null,
  };
};

const getUploadUrl = (fileMd5) => {
  return `${import.meta.env.VITE_BASE_FILE_API_URL}/file/upload/${fileMd5}`;
};
const onFileUploadSuccess = () => {
  message.success("文件上传成功");
  getMimeTypeOptions();
  getData();
};

const compressLoading = ref(false);
const onFilesCompress = () => {
  if (selectionFileMd5s.value.length < 2) {
    message.warning("至少需要选择两个文件");
  } else {
    compressLoading.value = true;
    CompressApi.compress(selectionFileMd5s.value)
      .then((res) => {
        compressLoading.value = false;
        window.open(res);
      })
      .catch(() => {
        compressLoading.value = false;
      });
  }
};

const referenceModalVisible = ref(false);
const referenceFileMd5 = ref(null);

const fileInfoModalVisible = ref(false);
const currentFileId = ref(null);
</script>

<template>
  <div class="file-view">
    <search-form
      v-model:keyword="searchModel.keyword"
      keyword-placeholder="请输入搜索关键字（文件名或md5）"
      :keyword-input-span="6"
      :on-clear="onClearSearchForm"
      :on-search="resetData"
    >
      <n-form-item-gi :span="4" label="后缀" path="mimeType">
        <n-input
          v-model:value="searchModel.extension"
          :maxlength="20"
          placeholder="请输入文件后缀"
          clearable
          show-count
        />
      </n-form-item-gi>
      <n-form-item-gi :span="5" label="类型" path="mimeType">
        <n-select
          v-model:value="searchModel.mimeType"
          :loading="mimeTypesLoading"
          :options="mimeTypeOptions"
          :render-option="renderToolTipOption"
          clearable
          filterable
          placeholder="请选择文件类型"
        />
      </n-form-item-gi>
      <n-form-item-gi :span="3" label="分类" path="type">
        <n-select
          v-model:value="searchModel.type"
          :disabled="searchModel.mimeType"
          :loading="fileTypeLoading"
          :options="fileTypeOptions"
          clearable
          placeholder="请选择文件分类"
        />
      </n-form-item-gi>
      <template #extra-button>
        <n-form-item-gi :span="3">
          <div class="display-inline-flex h-right">
            <file-upload
              :on-upload-success="onFileUploadSuccess"
              :upload-url="getUploadUrl"
              button-class="mr-10"
              style="width: unset"
            />
            <n-button
              :loading="compressLoading"
              secondary
              type="success"
              @click="onFilesCompress"
            >
              <template #icon>
                <n-icon>
                  <BriefcaseOutline />
                </n-icon>
              </template>
              打包下载
            </n-button>
          </div>
        </n-form-item-gi>
      </template>
    </search-form>
    <n-image-group>
      <n-data-table
        v-model:checked-row-keys="selectionFileMd5s"
        :max-height="630"
        :min-height="630"
        :columns="columns"
        :data="data"
        :loading="loading"
        :pagination="pagination"
        :row-key="(rowData) => rowData.md5"
        :single-line="false"
        remote
        :scroll-x="1600"
        @update:sorter="onTableSorterChange"
      />
    </n-image-group>
    <file-reference-table-modal
      v-model:show="referenceModalVisible"
      :file-md5="referenceFileMd5"
    />
    <file-info-form-modal
      v-model:show="fileInfoModalVisible"
      :file-id="currentFileId"
      :on-success="getData"
    />
  </div>
</template>
```
